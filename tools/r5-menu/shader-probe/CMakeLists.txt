cmake_minimum_required(VERSION 3.24)
set(CMAKE_C_FLAGS_INIT "--ld-path=\"${CMAKE_CURRENT_LIST_DIR}/../../../artifacts/xex-adapt/paso-b/toolbin/lld-link.exe\"")
set(CMAKE_CXX_FLAGS_INIT "${CMAKE_C_FLAGS_INIT}")
project(r5_primitive_shader_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../r4-shaders" r4 EXCLUDE_FROM_ALL)
get_target_property(probe_sources xenos-to-spirv SOURCES)
list(REMOVE_ITEM probe_sources xenos-to-spirv.cpp)
set(resolved_sources)
foreach(source IN LISTS probe_sources)
  if(IS_ABSOLUTE "${source}")
    list(APPEND resolved_sources "${source}")
  else()
    list(APPEND resolved_sources "${CMAKE_CURRENT_LIST_DIR}/../../r4-shaders/${source}")
  endif()
endforeach()
# Isolated correction: rectangle loop already terminates the then block by
# branching to its continuation. IfBuilder must not append a second terminator.
# Preserve the installed SDK and the original R4 translator unchanged.
set(rectangle_source "${REX_SDK}/src/graphics/pipeline/shader/spirv_translator.cpp")
file(READ "${rectangle_source}" rectangle_text)
string(REGEX MATCHALL "if_not_last_guest_vertex.makeBeginElse\\(\\)" occurrences "${rectangle_text}")
list(LENGTH occurrences occurrence_count)
if(NOT occurrence_count EQUAL 1)
  message(FATAL_ERROR "Rectangle probe correction no longer has one exact source match")
endif()
string(REPLACE "if_not_last_guest_vertex.makeBeginElse();" "if_not_last_guest_vertex.makeBeginElse(false);" rectangle_text "${rectangle_text}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/rectangle-translator.cpp" "${rectangle_text}")
list(REMOVE_ITEM resolved_sources "${rectangle_source}")
list(APPEND resolved_sources "${CMAKE_CURRENT_BINARY_DIR}/rectangle-translator.cpp")
add_executable(primitive-shader primitive-shader.cpp ${resolved_sources})
foreach(prop INCLUDE_DIRECTORIES COMPILE_DEFINITIONS LINK_LIBRARIES)
  get_target_property(value xenos-to-spirv ${prop})
  set_target_properties(primitive-shader PROPERTIES ${prop} "${value}")
endforeach()
add_custom_command(TARGET primitive-shader POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "${CMAKE_CURRENT_LIST_DIR}/../../../artifacts/xex-adapt/paso-b/src/rexglue-sdk-090/out/win-amd64/rexruntime.dll"
  "$<TARGET_FILE_DIR:primitive-shader>")
